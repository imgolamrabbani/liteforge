export interface ColumnSchema {
  cid: number;
  name: string;
  type: string;
  notNull: boolean;
  defaultValue: string | null;
  primaryKey: number; // 0 = not PK, 1..n = PK ordinal
}

export interface ForeignKeySchema {
  id: number;
  fromTable: string;
  fromColumn: string;
  toTable: string;
  toColumn: string;
  onUpdate: string;
  onDelete: string;
}

export interface IndexSchema {
  name: string;
  tableName: string;
  unique: boolean;
  columns: string[];
  sql: string | null;
}

export interface TableSchema {
  name: string;
  type: 'table' | 'view';
  sql: string;
  rowCount: number;
  columns: ColumnSchema[];
  foreignKeys: ForeignKeySchema[];
  indexes: IndexSchema[];
}

export interface PageQueryOptions {
  table: string;
  page: number;           // 1-based
  pageSize: number;       // Default 100, max 1000
  sortColumn?: string;
  sortDir?: 'ASC' | 'DESC';
  search?: string;        // Case-insensitive LIKE across text/numeric columns
}

export interface PageResult {
  table: string;
  columns: string[];
  rows: Array<{ rowid: number | null; values: CellValue[] }>;
  totalRows: number;
  page: number;
  pageSize: number;
  durationMs: number;
}

export interface SqlQueryResult {
  columns: string[];
  rows: CellValue[][];
  rowsAffected: number;
  durationMs: number;
  error?: string;

  kind: 'update' | 'insert' | 'delete';
  table: string;
  rowid?: number;
  column?: string;
  oldValue?: CellValue;
  newValue?: CellValue;
  rowValues?: Record<string, CellValue>;
}

export interface EditPreviewResult {

    columns: ColumnSchema[];

    | 'table_added'
    | 'table_removed'

  destructive: boolean;

## 5. WASM Engine, Safety & Atomic Persistence Design

1. **Zero-Native-Binary WASM Loading (`wasmEngine.ts`):**
   - Uses `sql.js` (`initSqlJs({ wasmBinary })`), reading `path.join(__dirname, 'sql-wasm.wasm')` via `node:fs/promises`.
   - `esbuild.config.mjs` copies `node_modules/sql.js/dist/sql-wasm.wasm` to `dist/sql-wasm.wasm` on every build.
   - All schema introspection queries use strict single-quoted string literals (`SELECT name, type, sql FROM sqlite_master WHERE type IN ('table', 'view') AND name NOT LIKE 'sqlite_%' ORDER BY name`) — completely immune to SQLite 3.41+ `DQS=0` double-quoted literal crashes!
   - All dynamic table/column identifiers are escaped via `quoteIdent(id) => '"' + id.replace(/"/g, '""') + '"'`; all cell values use bound `?` parameters.
2. **Transactional Inline Editing & Atomic Write-Back (`inlineEditor.ts` — Pro):**
   - Users edit cells, insert rows, or mark rows for deletion in a staged queue (`StagedEdit[]`).
   - Clicking **Preview & Apply** displays the exact SQL diff (`EditPreviewResult`) before touching disk.
   - Executes inside `BEGIN TRANSACTION; ...; COMMIT;` (calling `ROLLBACK;` on any constraint error so in-memory DB stays clean).
   - Exports `Uint8Array` via `db.export()`, writes to `<dbPath>.liteforge.tmp`, and atomically renames over `<dbPath>` via `fs.rename` so sudden crashes never leave a half-written database file.
3. **Visual Schema & ERD Inspector (`schemaErd.ts` — Pro):**
   - Inspects `PRAGMA table_info`, `PRAGMA foreign_key_list`, and `PRAGMA index_list` across all user tables.
   - Renders clean, deterministic auto-laid-out interactive table cards + FK edge connectors in the Webview (styled with `--vscode-*` CSS variables, zero external CDN dependencies).
4. **2-File SQLite Schema Diff & Migration SQL Generator (`dbDiff.ts` — Pro):**
   - Opens both source (baseline) and target (new) `.sqlite` files in isolated WASM instances.
   - Computes added/removed tables, added/removed/retyped columns, and added/removed indexes, emitting a ready-to-run transactional migration `.sql` script (marking destructive drops clearly with `-- WARNING: destructive` comments).
5. **Full `.sql` Dump & Multi-Table Export (`sqlDumper.ts` — Pro):**
   - Generates a complete `BEGIN TRANSACTION; ... CREATE TABLE ...; INSERT INTO ...; CREATE INDEX ...; COMMIT;` `.sql` dump with proper SQLite literal escaping (including `X'...'` hex literals for `BLOB` columns and `NULL`), plus a multi-table JSON bundle export.
